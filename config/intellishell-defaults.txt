# Quick start Claude Code (skip permissions) #claude #quick
claude --dangerously-skip-permissions

# Start Claude Code with debug logging #claude #debug
claude --dangerously-skip-permissions --debug

# Start Claude Code normally #claude
claude

# View latest debug log #claude #logs
cat ~/.claude/debug/latest

# Export session log to workspace #claude #logs
mkdir -p ~/.claude/exported-logs && cp ~/.claude/debug/latest ~/.claude/exported-logs/session-$(date +%Y%m%d-%H%M%S).log

# Log a run result #dev-log #logging
dev-log run "command" pass "description"

# Log a decision #dev-log #logging
dev-log decision "description of what was decided and why"

# Log an error #dev-log #logging
dev-log error "what failed" "root cause or detail"

# Log a change #dev-log #logging
dev-log change "what changed" "why it changed"

# View recent log entries #dev-log #logging
dev-log show
